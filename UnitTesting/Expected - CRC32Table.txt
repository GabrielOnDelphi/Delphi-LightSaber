# Rebuilds the standard reflected CRC-32 table (polynomial 0x04C11DB7, reflected form 0xEDB88320)
# from the definition, and the CRC-32 check value of b"123456789".
# Source of the algorithm: RFC 1952 section 8, make_crc_table().
import zlib

POLY = 0xEDB88320

table = []
for n in range(256):
    c = n
    for _ in range(8):
        c = (c >> 1) ^ POLY if c & 1 else c >> 1
    table.append(c)


def crc32_bitwise(data):
    crc = 0xFFFFFFFF
    for b in data:
        crc ^= b
        for _ in range(8):
            crc = (crc >> 1) ^ POLY if crc & 1 else crc >> 1
    return crc ^ 0xFFFFFFFF


msg = b"123456789"
bitwise = crc32_bitwise(msg)
zl = zlib.crc32(msg) & 0xFFFFFFFF
print("bitwise CRC-32(\"123456789\") = 0x%08X" % bitwise)
print("zlib.crc32(\"123456789\")     = 0x%08X" % zl)
print("match:", bitwise == zl == 0xCBF43926)

# Table-driven CRC, as a cross-check that the table itself is right
crc = 0xFFFFFFFF
for b in msg:
    crc = table[(crc ^ b) & 0xFF] ^ (crc >> 8)
print("table-driven CRC-32(\"123456789\") = 0x%08X" % (crc ^ 0xFFFFFFFF))

print("TABLE")
for i in range(0, 256, 8):
    line = ", ".join("$%08X" % v for v in table[i:i + 8])
    print("{ %3d } %s%s" % (i, line, "," if i < 248 else ""))
